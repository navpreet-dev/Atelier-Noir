    85→    85→export function MediaBg({
    86→    86→  src,
    87→    87→  poster,
    88→    88→  imgClassName = "",
    89→    89→  eager = false,
    90→    90→  videoClassName = "",
    91→    91→}: {
    92→    92→  src: string;
    93→    93→  poster: string;
    94→    94→  imgClassName?: string;
    95→    95→  eager?: boolean;
    96→    96→  videoClassName?: string;
    97→    97→}) {
    98→    98→  const ref = useRef<HTMLVideoElement>(null);
    99→    99→  const [ready, setReady] = useState(false);
   100→   100→  useEffect(() => {
   101→   101→    const v = ref.current;
   102→   102→    if (!v) return;
   103→   103→    const on = () => setReady(true);
   104→   104→    v.addEventListener("playing", on);
   105→   105→    return () => v.removeEventListener("playing", on);
   106→   106→  }, []);
   107→   107→  return (
   108→   108→    <>
   109→   109→      <video
   110→   110→        ref={ref}
   111→   111→        src={src}
   112→   112→        poster={poster}
   113→   113→        autoPlay
   114→   114→        muted
   115→   115→        loop
   116→   116→        playsInline
   117→   117→        preload={eager ? "auto" : "metadata"}
   118→   118→        disablePictureInPicture
   119→   119→        className={`absolute inset-0 h-full w-full object-cover transition-opacity duration-1000 ${videoClassName}`}
   120→   120→        style={{ opacity: ready ? 1 : 0 }}
   121→   121→      />
   122→   122→      <img
   123→   123→        src={poster}
   124→   124→        alt=""
   125→   125→        aria-hidden
   126→   126→        className={`pointer-events-none absolute inset-0 h-full w-full object-cover transition-opacity duration-1000 ${imgClassName}`}
   127→   127→        style={{ opacity: ready ? 0 : imgClassName.includes("opacity-") ? undefined : 1 }}
   128→   128→      />
   129→   129→    </>
   130→   130→  );
   131→   131→}

   170→   170→function ProductRail({

   198→   198→      live = false;
   199→   199→    };

   209→   209→        }`}

   211→   211→        <div className="mb-10 max-w-2xl text-left md:mb-14">
   212→   212→          <p className="mb-3 text-[11px] font-semibold uppercase tracking-[0.3em] text-[#B08968]">

   219→   219→        </div>
   220→   220→        {cta && layout === "grid" && (
   221→   221→          <button
   222→   222→            onClick={onCta}
   223→   223→            className="mb-10 hidden shrink-0 items-center gap-2 border-b border-stone-900 pb-1 text-[11px] font-semibold uppercase tracking-[0.25em] text-stone-900 hover:text-stone-500 md:flex"

   233→   233→                <div className="aspect-[3/4] animate-pulse bg-stone-100" />
   234→   234→              </div>

   602→   602→function FabricFilm() {
   603→   603→  const videoRef = useRef<HTMLVideoElement>(null);
   604→   604→  const [ending, setEnding] = useState(false);
   605→   605→  useEffect(() => {
   606→   606→    const v = videoRef.current;
   607→   607→    if (!v) return;
   608→   608→    const onTime = () => {
   609→   609→      if (!v.duration) return;
   610→   610→      const left = v.duration - v.currentTime;
   611→   611→      setEnding(left <= 3 && left > 0.05);
   612→   612→    };
   613→   613→    v.addEventListener("timeupdate", onTime);
   614→   614→    return () => v.removeEventListener("timeupdate", onTime);
   615→   615→  }, []);
   616→   616→  return (
   617→   617→    <section
   618→   618→      aria-label="Fabric detail film"
   619→   619→      className="relative h-[50vh] min-h-[380px] w-full overflow-hidden bg-stone-400"
   620→   620→    >
