    80→const formatCardNumber = (v: string) =>
    81→  v

    87→  const digits = v.replace(/\D/g, "").slice(0, 4);

    89→  return `${digits.slice(0, 2)}/${digits.slice(2)}`;

    95→  const cartItems = useStore((s) => s.cartItems);
    96→  const promoCode = useStore((s) => s.promoCode);
    97→  const sessionId = useStore((s) => s.sessionId);
    98→  const clearCart = useStore((s) => s.clearCart);
    99→  const requireSignIn = useStore((s) => s.requireSignIn);
   100→  const navigate = useStore((s) => s.navigate);
   101→
   102→  const [step, setStep] = useState<1 | 2 | 3>(1);
   103→  const [placing, setPlacing] = useState(false);

   117→  const [upiId, setUpiId] = useState("");
   118→  const [cardName, setCardName] = useState("");
   119→  const [cardNumber, setCardNumber] = useState("");
   120→  const [cardExpiry, setCardExpiry] = useState("");

   122→

   126→    setName((n) => n || user.name);

   130→
   131→  const subtotal = cartItems.reduce((n, i) => n + i.qty * i.product.price, 0);
   132→  const total = promoCode ? subtotal * 0.9 : subtotal;
   133→
   134→  if (!userReady) {
   135→    return <div className="min-h-[50vh]" aria-hidden />;

   200→      name: i.product.name,
   201→      image: i.product.images[0],
   202→      size: i.size,
   203→      color: i.color,
   204→      qty: i.qty,
   205→      price: i.product.price,
   206→    }));

   208→      paymentMethod === "card"

   229→          paymentDetail,
   230→          subtotal,
   231→          shipping: 0,
   232→          total,
   233→          sessionId,
   234→        }),
   235→      });

   237→        requireSignIn();
   238→        return;
   239→      }
   240→      const json = await res.json().catch(() => ({}));
   241→      if (!res.ok || !json.order) {
   242→        toast({
   243→          title: json.error ?? "Something went wrong placing your order.",
   244→          variant: "destructive",

   250→        firstName: (user.name ?? "").split(" ")[0] ?? "",

   252→        items,
   253→        name: name.trim() || user.name,
   254→        address: address.trim(),
   255→        city: city.trim(),
   256→        postalCode: postalCode.trim(),
   257→        country,
   258→        paymentDetail,
   259→        subtotal,
   260→        total,
   261→        promo: Boolean(promoCode),
   262→      });
   263→      await clearCart();
   264→      setStep(3);
   265→      scrollTop();
   266→      toast({ title: "Order placed — thank you." });
   267→    } catch {
   268→      toast({
   269→        title: "Something went wrong placing your order.",
   270→        variant: "destructive",
   271→      });
   272→    } finally {
   273→      setPlacing(false);
   274→    }
   275→  }
   276→

   278→    <div className="mx-auto max-w-7xl px-4 pb-24 pt-10 md:px-6 lg:px-10">
   279→      <h1 className="font-display text-3xl tracking-tight md:text-4xl">Checkout</h1>
   280→
   281→      {/* Step indicator */}

   283→        {STEPS.map((label, idx) => {
   284→          const n = idx + 1;
   285→          const cls =
   286→            n < step ? "text-[#5C6B4F]" : n === step ? "text-stone-900" : "text-stone-400";
   287→          return (
   288→            <span key={label} className={cls} aria-current={n === step ? "step" : undefined}>
   289→              {n < step ? `✓ ${label}` : `0${n} — ${label}`}
   290→            </span>
   291→          );
   292→        })}
   293→        <button
   294→          onClick={() => navigate({ name: "cart" })}
   295→          className="ml-auto text-[11px] font-semibold uppercase tracking-[0.2em] text-stone-500 underline transition hover:text-stone-900"
   296→        >
   297→          Return to Bag
   298→        </button>
   299→      </div>
   300→
   301→      <div className="mt-8 grid gap-12 lg:grid-cols-[1fr_380px]">
   302→        {/* Left — current step */}
   303→        <div>
   304→          {step === 1 ? (
   305→            <div className="space-y-8">
   306→              {/* Contact */}
   307→              <section className={SECTION_CLS}>
   308→                <h2 className={SECTION_TITLE_CLS}>Contact</h2>
   309→                <div className="space-y-5">
   310→                  <div className="space-y-2">
   311→                    <Label className={LABEL_CLS}>Full Name</Label>
   312→                    <Input
   313→                      value={name}
   314→                      onChange={(e) => setName(e.target.value)}
   315→                      placeholder="Jane Moreau"
   316→                      autoComplete="name"
   317→                      className={INPUT_CLS}
   318→                    />
   319→                  </div>
   320→                  <div className="space-y-2">
   321→                    <Label className={LABEL_CLS}>Email</Label>
   322→                    <Input
   323→                      type="email"
   324→                      value={email}
   325→                      onChange={(e) => setEmail(e.target.value)}
   326→                      placeholder="you@example.com"
   327→                      autoComplete="email"
   328→                      className={INPUT_CLS}
   329→                    />
   330→                  </div>
   331→                  <div className="space-y-2">
   332→                    <Label className={LABEL_CLS}>Phone</Label>
   333→                    <Input
   334→                      type="tel"
   335→                      value={phone}
   336→                      onChange={(e) => setPhone(e.target.value)}

   338→                      autoComplete="tel"
   339→                      className={INPUT_CLS}
   340→                    />
   341→                  </div>
   342→                </div>
   343→              </section>
   344→
   345→              {/* Shipping address */}

   347→                <h2 className={SECTION_TITLE_CLS}>Shipping Address</h2>

   356→                      className={INPUT_CLS}
   357→                    />
   358→                  </div>

   360→                    <div className="space-y-2">

   363→                        value={city}

   369→                    </div>

   379→                    </div>
   380→                  </div>

   382→                    <Label className={LABEL_CLS}>Country</Label>

   397→                    </Select>
   398→                  </div>
   399→                </div>
   400→              </section>
   401→
   402→              {/* Payment method */}
   403→              <section className={SECTION_CLS}>
   404→                <h2 className={SECTION_TITLE_CLS}>Payment Method</h2>

   407→                </p>
   408→                <RadioGroup
   409→                  value={paymentMethod}
   410→                  onValueChange={(v) => setPaymentMethod(v as PaymentMethod)}

   412→                >

   425→                        <p className="text-sm font-medium">{opt.title}</p>

   427→                      </div>
   428→                    </div>
   429→                  ))}
   430→                </RadioGroup>
   431→                {paymentMethod === "upi" && (
   432→                  <div className="space-y-2">
   433→                    <Label className={LABEL_CLS}>UPI ID</Label>
   434→                    <Input
   435→                      value={upiId}
   436→                      onChange={(e) => setUpiId(e.target.value)}
   437→                      placeholder="yourname@upi"
   438→                      className={INPUT_CLS}
   439→                    />
   440→                  </div>
   441→                )}
   442→              </section>
   443→
   444→              <button
   445→                onClick={handleContinueToPayment}
   446→                className="h-12 w-full bg-stone-900 text-[11px] font-semibold uppercase tracking-[0.25em] text-white transition hover:bg-stone-700"
   447→              >
   448→                Continue to Payment
   449→              </button>
   450→            </div>
   451→          ) : (
   452→            <div className="space-y-8">
   453→              <section className={SECTION_CLS}>
   454→                <h2 className={SECTION_TITLE_CLS}>
   455→                  {paymentMethod === "card"
   456→                    ? "Card Details"
   457→                    : paymentMethod === "upi"
   458→                      ? "UPI Details"
   459→                      : "Cash on Delivery"}
   460→                </h2>
   461→                {paymentMethod === "card" && (
   462→                  <p className="text-sm text-stone-500">
   463→                    Your card details are encrypted end to end.
   464→                  </p>
   465→                )}
   466→                {paymentMethod === "cod" && (
   467→                  <p className="text-sm text-stone-500">
   468→                    Pay in cash when your order arrives.
   469→                  </p>
   470→                )}
   471→
   472→                {paymentMethod === "card" && (
   473→                  <div className="space-y-5">
   474→                    <div className="space-y-2">
   475→                      <Label className={LABEL_CLS}>Name on Card</Label>
   476→                      <Input
   477→                        value={cardName}
   478→                        onChange={(e) => setCardName(e.target.value)}
   479→                        placeholder="JANE MOREAU"
   480→                        autoComplete="cc-name"
   481→                        className={INPUT_CLS}
   482→                      />
   483→                    </div>
   484→                    <div className="space-y-2">
   485→                      <Label className={LABEL_CLS}>Card Number</Label>
   486→                      <Input
   487→                        value={cardNumber}
   488→                        onChange={(e) => setCardNumber(formatCardNumber(e.target.value))}
   489→                        placeholder="4242 4242 4242 4242"
   490→                        inputMode="numeric"
   491→                        maxLength={19}
   492→                        autoComplete="cc-number"
   493→                        className={`${INPUT_CLS} tracking-[0.08em]`}
   494→                      />
   495→                    </div>
   496→                    <div className="grid gap-5 sm:grid-cols-2">
   497→                      <div className="space-y-2">
   498→                        <Label className={LABEL_CLS}>Expiry (MM/YY)</Label>
   499→                        <Input
   500→                          value={cardExpiry}
   501→                          onChange={(e) => setCardExpiry(formatExpiry(e.target.value))}
   502→                          placeholder="08/27"
   503→                          inputMode="numeric"
   504→                          maxLength={5}
   505→                          autoComplete="cc-exp"
   506→                          className={INPUT_CLS}
   507→                        />
   508→                      </div>
   509→                      <div className="space-y-2">
   510→                        <Label className={LABEL_CLS}>Security Code</Label>
   511→                        <Input
   512→                          type="password"
   513→                          value={cardCvv}
   514→                          onChange={(e) => setCardCvv(e.target.value.replace(/\D/g, "").slice(0, 4))}
   515→                          placeholder="3–4 digits"
   516→                          inputMode="numeric"
   517→                          maxLength={4}
   518→                          autoComplete="cc-csc"
   519→                          className={INPUT_CLS}
   520→                        />
   521→                      </div>
   522→                    </div>
   523→                  </div>
   524→                )}
   525→
   526→                {paymentMethod === "upi" && (
   527→                  <div className="space-y-2">
   528→                    <Label className={LABEL_CLS}>UPI ID</Label>
   529→                    <Input
   530→                      value={upiId}
   531→                      onChange={(e) => setUpiId(e.target.value)}
   532→                      placeholder="yourname@upi"
   533→                      className={INPUT_CLS}
   534→                    />
   535→                  </div>
   536→                )}
   537→              </section>
   538→
   539→              <div className="flex flex-col gap-3 sm:flex-row">
   540→                <button
   541→                  onClick={() => {
   542→                    setStep(1);
   543→                    scrollTop();
   544→                  }}
   545→                  className="h-12 flex-1 border border-stone-300 bg-transparent text-[11px] font-semibold uppercase tracking-[0.25em] text-stone-900 transition hover:border-stone-900"
   546→                >
   547→                  Return to Shipping
   548→                </button>
   549→                <button
   550→                  onClick={handlePlaceOrder}
   551→                  disabled={placing}
   552→                  className="h-12 flex-1 bg-stone-900 text-[11px] font-semibold uppercase tracking-[0.25em] text-white transition hover:bg-stone-700 disabled:opacity-60"
   553→                >
   554→                  {placing ? "Placing Order…" : `Place Order — ${formatPrice(total)}`}
   555→                </button>
   556→              </div>
   557→            </div>
   558→          )}
   559→        </div>
   560→
   561→        {/* Right — read-only order summary */}
   562→        <SummaryAside items={cartItems} subtotal={subtotal} total={total} promo={promoCode} />
   563→      </div>
   564→    </div>
   565→  );
   566→}

   568→/* ---------- Order summary (read-only, checkout) ---------- */
   569→
   570→function SummaryAside({
   571→  items,
   572→  subtotal,
   573→  total,
   574→  promo,
   575→}: {
   576→  items: { id: string; qty: number; size: string; color: string; product: { id: string; name: string; images: string[]; price: number } }[];
   577→  subtotal: number;
   578→  total: number;
   579→  promo: string | null;
   580→}) {
   581→  return (
   582→    <aside className="h-fit border border-stone-200 p-6 lg:sticky lg:top-24">
   583→      <h2 className="font-display text-lg tracking-[0.1em]">ORDER SUMMARY</h2>
   584→      <div className="thin-scrollbar mt-6 max-h-72 space-y-5 overflow-y-auto pr-1">

   747→/* ---------- Auth gate (sign in / create account) ---------- */
   748→
   749→function AuthPanel() {
   750→  const setUser = useStore((s) => s.setUser);
   751→  const navigate = useStore((s) => s.navigate);
   752→
   753→  const [mode, setMode] = useState<"login" | "register">("login");
   754→  const [name, setName] = useState("");
   755→  const [email, setEmail] = useState("");
   756→  const [password, setPassword] = useState("");
   757→  const [error, setError] = useState<string | null>(null);
   758→  const [busy, setBusy] = useState(false);
   759→
   760→  async function handleSubmit(e: FormEvent<HTMLFormElement>) {
   761→    e.preventDefault();
   762→    if (busy) return;
   763→    setError(null);
   764→    if (!email.trim() || !password || (mode === "register" && !name.trim())) {
   765→      setError("Please fill in all fields.");
   766→      return;
   767→    }
   768→    setBusy(true);
   769→    try {
   770→      const res = await fetch(mode === "login" ? "/api/auth/login" : "/api/auth/register", {
   771→        method: "POST",
   772→        headers: { "Content-Type": "application/json" },
   773→        body: JSON.stringify(
   774→          mode === "login"
   775→            ? { email: email.trim(), password }
   776→            : { name: name.trim(), email: email.trim(), password }
   777→        ),
   778→      });
   779→      const json = await res.json().catch(() => ({}));
   780→      if (!res.ok || !json.user) {
   781→        setError(json.error ?? "Something went wrong. Please try again.");
   782→        return;
   783→      }
   784→      setUser(json.user);
   785→      const first = String(json.user.name ?? "").split(" ")[0] ?? "";
   786→      toast({
   787→        title:
   788→          mode === "login" ? `Welcome back, ${first}.` : `Welcome to Atelier Noir, ${first}.`,
   789→      });
   790→      const pending = useStore.getState().pendingRoute;
   791→      if (pending) {
   792→        useStore.setState({ pendingRoute: null });
   793→        navigate(pending);
   794→      }
   795→    } catch {
   796→      setError("Something went wrong. Please try again.");
   797→    } finally {
   798→      setBusy(false);
   799→    }
   800→  }
   801→
   802→  return (
   803→    <div className="mx-auto max-w-md px-4 pb-24 pt-14">
   804→      <p className="text-center text-[10px] font-semibold uppercase tracking-[0.3em] text-stone-400">
   805→        Atelier Noir
   806→      </p>
   807→      <h1 className="mt-3 text-center font-display text-3xl md:text-4xl">
   808→        {mode === "login" ? "Sign In" : "Create Account"}
   809→      </h1>
   810→      <p className="mt-3 text-center text-sm text-stone-500">
   811→        {mode === "login"
   812→          ? "Welcome back. Sign in to see your saved activity, wishlist and bag."
   813→          : "One account for everything — your recently viewed pieces, wishlist and bag are kept in one place."}
   814→      </p>
   815→
   816→      <div className="mt-8 grid grid-cols-2 border border-stone-200">
   817→        <button
   818→          type="button"
   819→          onClick={() => {
   820→            setMode("login");
   821→            setError(null);
   822→          }}
   823→          className={`py-3 text-[11px] font-semibold uppercase tracking-[0.2em] transition ${
   824→            mode === "login" ? "bg-stone-900 text-white" : "text-stone-600 hover:text-stone-900"
   825→          }`}
   826→        >
   827→          Sign In
   828→        </button>
   829→        <button
   830→          type="button"
   831→          onClick={() => {
   832→            setMode("register");
   833→            setError(null);
   834→          }}
   835→          className={`py-3 text-[11px] font-semibold uppercase tracking-[0.2em] transition ${
   836→            mode === "register" ? "bg-stone-900 text-white" : "text-stone-600 hover:text-stone-900"
   837→          }`}
   838→        >
   839→          Create Account
   840→        </button>
   841→      </div>
   842→
   843→      <form onSubmit={handleSubmit} className="mt-8 space-y-5" noValidate>
   844→        {mode === "register" && (
   845→          <div className="space-y-2">
   846→            <Label className={LABEL_CLS}>Full Name</Label>
   847→            <Input
   848→              value={name}
   849→              onChange={(e) => setName(e.target.value)}
   850→              placeholder="Jane Moreau"
   851→              autoComplete="name"
   852→              className={INPUT_CLS}
   853→            />
   854→          </div>
   855→        )}
   856→        <div className="space-y-2">
   857→          <Label className={LABEL_CLS}>Email</Label>
   858→          <Input
   859→            type="email"
   860→            value={email}
   861→            onChange={(e) => setEmail(e.target.value)}
   862→            placeholder="you@example.com"
   863→            autoComplete="email"
   864→            className={INPUT_CLS}
   865→          />
   866→        </div>
   867→        <div className="space-y-2">
   868→          <Label className={LABEL_CLS}>Password</Label>
   869→          <Input
   870→            type="password"
   871→            value={password}
   872→            onChange={(e) => setPassword(e.target.value)}
   873→            placeholder={mode === "register" ? "At least 8 characters" : "Your password"}
   874→            autoComplete={mode === "register" ? "new-password" : "current-password"}
   875→            className={INPUT_CLS}
   876→          />
   877→        </div>
   878→        <button
   879→          type="submit"
   880→          disabled={busy}
   881→          className="h-12 w-full bg-stone-900 text-[11px] font-semibold uppercase tracking-[0.25em] text-white transition hover:bg-stone-700 disabled:opacity-60"
   882→        >
   883→          {busy ? "Please wait…" : mode === "login" ? "Sign In" : "Create Account"}
   884→        </button>
   885→        {error && <p className="mt-4 text-center text-sm text-[#9A3B26]">{error}</p>}
   886→      </form>
   887→
   888→      <p className="mt-6 text-center text-xs text-stone-400">
   889→        Your email and password are kept private and are never shared.
   890→      </p>
   891→    </div>
   892→  );
   893→}
