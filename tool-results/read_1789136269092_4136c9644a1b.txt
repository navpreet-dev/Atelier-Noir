    85→export function MediaBg({
    86→  src,
    87→  poster,
    88→  imgClassName = "",
    89→  eager = false,
    90→  videoClassName = "",
    91→}: {
    92→  src: string;
    93→  poster: string;
    94→  imgClassName?: string;
    95→  eager?: boolean;
    96→  videoClassName?: string;
    97→}) {
    98→  const ref = useRef<HTMLVideoElement>(null);
    99→  const [ready, setReady] = useState(false);
   100→  useEffect(() => {
   101→    const v = ref.current;
   102→    if (!v) return;
   103→    const on = () => setReady(true);
   104→    v.addEventListener("playing", on);
   105→    return () => v.removeEventListener("playing", on);
   106→  }, []);
   107→  return (
   108→    <>
   109→      <video
   110→        ref={ref}
   111→        src={src}
   112→        poster={poster}
   113→        autoPlay
   114→        muted
   115→        loop
   116→        playsInline
   117→        preload={eager ? "auto" : "metadata"}
   118→        disablePictureInPicture
   119→        className={`absolute inset-0 h-full w-full object-cover transition-opacity duration-1000 ${videoClassName}`}
   120→        style={{ opacity: ready ? 1 : 0 }}
   121→      />
   122→      <img
   123→        src={poster}
   124→        alt=""
   125→        aria-hidden
   126→        className={`pointer-events-none absolute inset-0 h-full w-full object-cover transition-opacity duration-1000 ${imgClassName}`}
   127→        style={{ opacity: ready ? 0 : imgClassName.includes("opacity-") ? undefined : 1 }}
   128→      />
   129→    </>
   130→  );
   131→}

   170→function ProductRail({
   171→  eyebrow,
   172→  title,
   173→  subtitle,
   174→  cta,
   175→  onCta,
   176→  query,
   177→  layout,
   178→}: {
   179→  eyebrow: string;
   180→  title: string;
   181→  subtitle: string;
   182→  cta?: string;
   183→  onCta?: () => void;
   184→  query: string;
   185→  layout: "grid" | "bestsellers";
   186→}) {
   187→  const [products, setProducts] = useState<Product[] | null>(null);
   188→
   189→  useEffect(() => {
   190→    let live = true;
   191→    fetch(query)
   192→      .then((r) => r.json())
   193→      .then((d) => {
   194→        if (live) setProducts(d.products ?? []);
   195→      })
   196→      .catch(() => live && setProducts([]));
   197→    return () => {
   198→      live = false;
   199→    };
   200→  }, [query]);
   201→
   202→  return (
   203→    <section className="mx-auto max-w-7xl px-4 pb-20 md:px-6 lg:px-10">
   204→      <div
   205→        className={`mb-10 ${
   206→          layout === "grid"
   207→            ? "flex items-end justify-between"
   208→            : "flex flex-col gap-6 md:flex-row md:items-end md:justify-between"
   209→        }`}
   210→      >
   211→        <div className="mb-10 max-w-2xl text-left md:mb-14">
   212→          <p className="mb-3 text-[11px] font-semibold uppercase tracking-[0.3em] text-[#B08968]">
   213→            {eyebrow}
   214→          </p>
   215→          <h2 className="font-display text-3xl leading-tight tracking-tight text-stone-900 md:text-[2.75rem]">
   216→            {title}
   217→          </h2>
   218→          <p className="mt-4 text-sm leading-relaxed text-stone-500 md:text-[15px]">{subtitle}</p>
   219→        </div>
   220→        {cta && layout === "grid" && (
   221→          <button
   222→            onClick={onCta}
   223→            className="mb-10 hidden shrink-0 items-center gap-2 border-b border-stone-900 pb-1 text-[11px] font-semibold uppercase tracking-[0.25em] text-stone-900 hover:text-stone-500 md:flex"
   224→          >
   225→            {cta} <ArrowRight size={14} aria-hidden />
   226→          </button>
   227→        )}
   228→      </div>
   229→      <div className={layout === "grid" ? "grid grid-cols-2 gap-6 md:grid-cols-4" : "grid grid-cols-2 gap-x-4 gap-y-10 md:grid-cols-4 md:gap-x-6"}>
   230→        {products === null
   231→          ? [0, 1, 2, 3].map((i) => (
   232→              <div key={i}>
   233→                <div className="aspect-[3/4] animate-pulse bg-stone-100" />
   234→              </div>
   235→            ))
   236→          : products.map((p, i) => <ProductCard key={p.id} product={p} eager={i < 4} />)}
   237→      </div>
   238→      {cta && layout === "bestsellers" && (
   239→        <div className="mt-12 text-center">
   240→          <button
   241→            onClick={onCta}
   242→            className="border border-stone-900 px-10 py-4 text-[11px] font-semibold uppercase tracking-[0.25em] text-stone-900 transition hover:bg-stone-900 hover:text-white"
   243→          >
   244→            {cta}
   245→          </button>
   246→        </div>
   247→      )}
   248→    </section>
   249→  );
   250→}

   602→function FabricFilm() {
   603→  const videoRef = useRef<HTMLVideoElement>(null);
   604→  const [ending, setEnding] = useState(false);
   605→  useEffect(() => {
   606→    const v = videoRef.current;
   607→    if (!v) return;
   608→    const onTime = () => {
   609→      if (!v.duration) return;
   610→      const left = v.duration - v.currentTime;
   611→      setEnding(left <= 3 && left > 0.05);
   612→    };
   613→    v.addEventListener("timeupdate", onTime);
   614→    return () => v.removeEventListener("timeupdate", onTime);
   615→  }, []);
   616→  return (
   617→    <section
   618→      aria-label="Fabric detail film"
   619→      className="relative h-[50vh] min-h-[380px] w-full overflow-hidden bg-stone-400"
   620→    >
   621→      <video
   622→        ref={videoRef}
   623→        src="/videos/fabric-detail.mp4"
   624→        poster="/images/fabric/fabric-4.jpg"
   625→        autoPlay
   626→        muted
   627→        loop
   628→        playsInline
   629→        preload="auto"
   630→        disablePictureInPicture
   631→        className="absolute inset-0 h-full w-full object-cover"
   632→      />
   633→      <div
   634→        aria-hidden={!ending}
   635→        className={`pointer-events-none absolute inset-0 flex items-center justify-center bg-stone-950/30 transition-opacity duration-1000 ${
   636→          ending ? "opacity-100" : "opacity-0"
   637→        }`}
   638→      >
   639→        <p className="font-display text-4xl tracking-[0.4em] text-white md:text-6xl">ATELIER</p>
   640→      </div>
   641→    </section>
   642→  );
   643→}
