   750→   750→  const setUser = useStore((s) => s.setUser);
   751→   751→  const navigate = useStore((s) => s.navigate);
   752→   752→
   753→   753→  const [mode, setMode] = useState<"login" | "register">("login");
   754→   754→  const [name, setName] = useState("");
   755→   755→  const [email, setEmail] = useState("");
   756→   756→  const [password, setPassword] = useState("");
   757→   757→  const [error, setError] = useState<string | null>(null);
   758→   758→  const [busy, setBusy] = useState(false);
   759→   759→
   760→   760→  async function handleSubmit(e: FormEvent<HTMLFormElement>) {
   761→   761→    e.preventDefault();
   762→   762→    if (busy) return;
   763→   763→    setError(null);

   770→   770→      const res = await fetch(mode === "login" ? "/api/auth/login" : "/api/auth/register", {
   771→   771→        method: "POST",
   772→   772→        headers: { "Content-Type": "application/json" },
   773→   773→        body: JSON.stringify(
   774→   774→          mode === "login"
   775→   775→            ? { email: email.trim(), password }
   776→   776→            : { name: name.trim(), email: email.trim(), password }
   777→   777→        ),
   778→   778→      });
   779→   779→      const json = await res.json().catch(() => ({}));
   780→   780→      if (!res.ok || !json.user) {
   781→   781→        setError(json.error ?? "Something went wrong. Please try again.");
   782→   782→        return;

   784→   784→      setUser(json.user);

   788→   788→          mode === "login" ? `Welcome back, ${first}.` : `Welcome to Atelier Noir, ${first}.`,

   803→   803→    <div className="mx-auto max-w-md px-4 pb-24 pt-14">
